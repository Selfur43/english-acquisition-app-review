final class AuthenticationController extends ChangeNotifier {
  AuthenticationController(this._repository);

  final AuthenticationRepository _repository;

  String? errorMessage;
  bool isBusy = false;
  AuthenticationStatus status = AuthenticationStatus.initializing;
  AuthenticatedUser? user;

  Future<void> initialize() async {
    user = await _repository.restore();
    status = user == null
        ? AuthenticationStatus.unauthenticated
        : AuthenticationStatus.authenticated;
    notifyListeners();
  }

  Future<void> signIn({
    required String password,
    required String userName,
  }) async {
    errorMessage = null;
    isBusy = true;
    notifyListeners();
    try {
      user = await _repository.signIn(
        password: password,
        userName: userName,
      );
      status = AuthenticationStatus.authenticated;
    } on AuthenticationFailure catch (failure) {
      errorMessage = failure.message;
      status = AuthenticationStatus.unauthenticated;
    } finally {
      isBusy = false;
      notifyListeners();
    }
  }

  Future<void> signOut() async {
    await _repository.signOut();
    errorMessage = null;
    status = AuthenticationStatus.unauthenticated;
    user = null;
    notifyListeners();
  }
}

import '../../domain/entities/authenticated_user.dart';
import '../../domain/repositories/authentication_repository.dart';
import '../secure_storage/credential_store.dart';

final class AuthenticationFailure implements Exception {
  const AuthenticationFailure(this.message);

  final String message;
}

final class AuthenticationRepositoryAdapter
    implements AuthenticationRepository {
  const AuthenticationRepositoryAdapter(this._credentialStore);

  final CredentialStore _credentialStore;

  static const _users = <String, AuthenticatedUser>{
    'furkan': AuthenticatedUser(displayName: 'Furkan', id: 'learner-1'),
    'learner2': AuthenticatedUser(displayName: 'Learner 2', id: 'learner-2'),
  };

  @override
  Future<AuthenticatedUser?> restore() async {
    final userId = await _credentialStore.readUserId();
    for (final user in _users.values) {
      if (user.id == userId) return user;
    }
    return null;
  }

  @override
  Future<AuthenticatedUser> signIn({
    required String password,
    required String userName,
  }) async {
    final user = _users[userName.trim().toLowerCase()];
    if (user == null || password != 'change-me-now') {
      throw const AuthenticationFailure('Invalid credentials.');
    }
    await _credentialStore.writeUserId(user.id);
    return user;
  }

  @override
  Future<void> signOut() => _credentialStore.clear();
}

final class _SignInPageState extends ConsumerState<SignInPage> {
  final _passwordController = TextEditingController();
  final _userNameController = TextEditingController();

  @override
  void dispose() {
    _passwordController.dispose();
    _userNameController.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final authentication = ref.watch(authenticationControllerProvider);
    return Scaffold(
      body: SafeArea(
        child: Center(
          child: SingleChildScrollView(
            padding: const EdgeInsets.all(24),
            child: ConstrainedBox(
              constraints: const BoxConstraints(maxWidth: 420),
              child: Column(
                mainAxisSize: MainAxisSize.min,
                children: [
                  Text(
                    'English Acquisition',
                    style: Theme.of(context).textTheme.headlineMedium,
                  ),
                  const SizedBox(height: 8),
                  const Text('Private two-user learning application'),
                  const SizedBox(height: 32),
                  TextField(
                    controller: _userNameController,
                    decoration: const InputDecoration(
                      border: OutlineInputBorder(),
                      labelText: 'User name',
                    ),
                    textInputAction: TextInputAction.next,
                  ),
                  const SizedBox(height: 16),
                  TextField(
                    controller: _passwordController,
                    decoration: const InputDecoration(
                      border: OutlineInputBorder(),
                      labelText: 'Password',
                    ),
                    obscureText: true,
                    onSubmitted: (_) => _submit(),
                  ),
                  if (authentication.errorMessage != null)
                    Padding(
                      padding: const EdgeInsets.only(top: 12),
                      child: Text(
                        authentication.errorMessage!,
                        style: TextStyle(
                          color: Theme.of(context).colorScheme.error,
                        ),
                      ),
                    ),
                  const SizedBox(height: 20),
                  SizedBox(
                    width: double.infinity,
                    child: FilledButton(
                      onPressed: authentication.isBusy ? null : _submit,
                      child: Text(
                        authentication.isBusy ? 'Signing in...' : 'Sign in',
                      ),
                    ),
                  ),
                ],
              ),
            ),
          ),
        ),
      ),
    );
  }

  Future<void> _submit() =>
      ref.read(authenticationControllerProvider).signIn(
            password: _passwordController.text,
            userName: _userNameController.text,
          );
